34// Backend that outputs logs to stdout.
35class StdoutLogBackend final : public BaseLogBackend {
36public:
37 StdoutLogBackend() = default;
38
39 ~StdoutLogBackend() {
40 // Safe cleanup in destructor
41 try {
42 if (IsInitialized()) {
43 DoFinalize();
44 }
45 } catch (...) {
46 // Suppress exceptions in destructor
47 }
48 }
49
50protected:
51 std::string name_ = "";
52 // In STDOUT_BACKEND, DoInitialize() just stores the name
53 void DoInitialize(const char* name) override {
54 name_ = name ? name : "";
55 }
56
57 // In STDOUT_BACKEND, DoFinalize() does nothing
58 void DoFinalize() override {
59 name_ = "";
60 }
61
62 // DoOutput received log as a single line to stdout
63 // Format: yyyy-MM-ddTHH:mm:ss.SSSZ,PID,TID,LEVEL,name,TAG,msg
64 void DoOutput(std::string_view tag, std::string_view msg, LogLevel level) override {
65 // Do not output logs below the minimum log level
66 if (level > log_level_threshold_) return;
67
68 // Initialization check is performed by BaseLogBackend::Output before calling DoOutput.
69 std::cout << Timestamp() << ',';
70 std::cout << getpid() << "," << std::this_thread::get_id() << "," << LevelToString(level)
71 << "," << name_ << "," << tag << "," << msg << std::endl;
72 }
73
74 void DoSetLogLevel(LogLevel level) override {
75 log_level_threshold_ = level;
76 }
77
78private:
79 // Convert LogLevel to string
80 static const char* LevelToString(LogLevel level) {
81 switch (level) {
82 case LogLevel::kFatal:
83 return "FATAL";
84 case LogLevel::kError:
85 return "ERROR";
86 case LogLevel::kWarn:
87 return "WARN";
88 case LogLevel::kInfo:
89 return "INFO";
90 case LogLevel::kDebug:
91 return "DEBUG";
92 default:
93 return "UNKNOWN";
94 }
95 }
96
97 // Simple timestamp (utc time): "yyyy-MM-ddTHH:mm:ss.SSSZ"
98 std::string Timestamp() const {
99 auto now = std::chrono::system_clock::now();
100 std::time_t epochtime = std::chrono::system_clock::to_time_t(now);
101 auto ms = std::chrono::duration_cast<std::chrono::milliseconds>(
102 now.time_since_epoch()) % 1000;
103
104 std::tm tm{};
105 gmtime_r(&epochtime, &tm);
106
107 char buffer[sizeof("yyyy-MM-ddTHH:mm:ss.SSSZ")];
108 int len =
109 snprintf(buffer, sizeof(buffer), "%04d-%02d-%02dT%02d:%02d:%02d.%03dZ",
110 tm.tm_year + 1900, tm.tm_mon + 1, tm.tm_mday, tm.tm_hour,
111 tm.tm_min, tm.tm_sec, static_cast<int>(ms.count()));
112 return std::string(buffer, len);
113 }
114
115};
