204template <typename U, typename Variant>
205struct IsVariantAlternative : std::false_type {};
206
207template <typename U, typename... Ts>
208struct IsVariantAlternative<U, std::variant<Ts...>>
209 : std::disjunction<std::is_same<U, Ts>...> {};
210
211template <typename U, typename Variant>
212inline constexpr bool IsVariantAlternative_v =
213 IsVariantAlternative<U, Variant>::value;
214
229template <typename U>
230inline std::optional<U> ExtractValue(const yoriito::ValueType& value) {
231 if constexpr (IsVariantAlternative_v<U, yoriito::ValueType>) {
232 if (std::holds_alternative<U>(value)) {
233 return std::get<U>(value);
234 }
235 }
236 return std::nullopt;
237}
