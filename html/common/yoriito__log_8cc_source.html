35void Log::Initialize(const char* name) {

37 if (!BackendManager().backend_) {

41 return; // Already initialized

44}

46void Log::Finalize() {

48 if (!BackendManager().backend_) {
49 return;
50 }

52 return;
53 }
54 BackendManager().backend_->Finalize();
