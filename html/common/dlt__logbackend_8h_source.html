34constexpr int kDltIdMaxLength = 4; // Max length for DLT IDs
35constexpr std::string_view kDefaultDltID = "----"; // Default DLT ID
36
37// Backend that outputs logs to DLT.
38class DltLogBackend final : public BaseLogBackend {
39public:
40 DltLogBackend() = default;
41 ~DltLogBackend() {
42 // Safe cleanup in destructor
43 try {
44 // Avoid finalization if not initialized
45 if (IsInitialized()) {
46 DoFinalize();
47 }
48 } catch (...) {
49 // Suppress exceptions in destructor
50 }
51 }
52protected:
53 std::string name_ = "";
54 void DoInitialize(const char* name) override {
55 name_ = name ? std::string(name) : "";
56 // Adjusts the APID to fit DLT's size restrictions
57 std::string standardized_app_id = ConvertToDltId(name_);
58 DLT_REGISTER_APP(standardized_app_id.c_str(), name_.c_str());
59 }
60
61 void DoFinalize() override {
62 name_ = "";
63 for (auto& context_entry : context_map_) {
64 DLT_UNREGISTER_CONTEXT(*(context_entry.second));
65 }
66 context_map_.clear();
67 DLT_UNREGISTER_APP();
68 }
69
70 void DoOutput(std::string_view tag, std::string_view msg, LogLevel level) override {
71 DltContext& context = EnsureContext(std::string(tag));
72
73 pid_t pid = getpid();
74 std::ostringstream tid_stream;
75 tid_stream << std::this_thread::get_id();
76 std::string tid_str = tid_stream.str();
77
78 DLT_LOG(context, ConvertToDltLevel(level),
79 DLT_UINT32(pid),
80 DLT_CSTRING(tid_str.c_str()),
81 DLT_CSTRING(name_.c_str()),
82 DLT_SIZED_CSTRING(tag.data(), tag.size()),
83 DLT_SIZED_CSTRING(msg.data(), msg.size()));
84 }
85
86 void DoSetLogLevel(LogLevel level) override {
87 log_level_threshold_ = level;
88 DLT_SET_APPLICATION_LL_TS_LIMIT(ConvertToDltLevel(level), DLT_TRACE_STATUS_OFF);
89 }
90private:
91 // Convert LogLevel to DltLogLevelType
92 static DltLogLevelType ConvertToDltLevel(LogLevel level) {
93 switch (level) {
94 case LogLevel::kFatal: return DLT_LOG_FATAL;
95 case LogLevel::kError: return DLT_LOG_ERROR;
96 case LogLevel::kWarn: return DLT_LOG_WARN;
97 case LogLevel::kInfo: return DLT_LOG_INFO;
98 case LogLevel::kDebug: return DLT_LOG_DEBUG;
99 default: return DLT_LOG_INFO;
100 }
101 }
102
103 // DLT requires context ID and APID to be max characters
104 static std::string ConvertToDltId(const std::string& input_id) {
105 // Return default ID for empty string as-is
106 if (input_id.empty()) return std::string(kDefaultDltID);
107 if (input_id.size() <= kDltIdMaxLength) return input_id;
108 return input_id.substr(0, kDltIdMaxLength);
109 }
110
111 DltContext& EnsureContext(const std::string& tag) {
112 const std::string context_id = ConvertToDltId(tag);
113 auto it = context_map_.find(context_id);
114 if (it != context_map_.end()) return *(it->second);
115
116 // Insert before registration to ensure cleanup on exception
117 auto context = std::make_unique<DltContext>();
118 auto [iter, inserted] = context_map_.try_emplace(context_id, std::move(context));
119 if (!inserted) return *(iter->second);
120
121 DLT_REGISTER_CONTEXT_LL_TS(*(iter->second), context_id.c_str(), tag.c_str(), ConvertToDltLevel(log_level_threshold_), DLT_TRACE_STATUS_OFF); // Full tag in description
122 return *(iter->second);
123 }
124
125 std::unordered_map<std::string, std::unique_ptr<DltContext>> context_map_;
126};
127
