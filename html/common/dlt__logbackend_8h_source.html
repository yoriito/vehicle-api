39// Backend that outputs logs to DLT.
40class DltLogBackend final : public BaseLogBackend {
41 public:
42 DltLogBackend() = default;
43 ~DltLogBackend() noexcept override {
44 // Safe cleanup in destructor
45 try {
46 // Avoid finalization if not initialized
47 if (BaseLogBackend::IsInitialized()) {
48 CleanupDltResources();
49 }
50 } catch (...) {
51 // Suppress exceptions in destructor
52 }
53 }
54 DltLogBackend(const DltLogBackend&) = delete;
55 DltLogBackend& operator=(const DltLogBackend&) = delete;
56 DltLogBackend(DltLogBackend&&) = delete;
57 DltLogBackend& operator=(DltLogBackend&&) = delete;
58
59 protected:
60 void DoInitialize(const char* name) override {
61 name_ = name ? std::string(name) : "";
62 // Adjusts the APID to fit DLT's size restrictions
63 std::string standardized_app_id = ConvertToDltId(name_);
64 DLT_REGISTER_APP(standardized_app_id.c_str(), name_.c_str());
65 }
66
67 void DoFinalize() override { CleanupDltResources(); }
68
69 void DoOutput(std::string_view tag, std::string_view msg,
70 LogLevel level) override {
71 DltContext& context = EnsureContext(std::string(tag));
72
73 pid_t pid = getpid();
74 std::ostringstream tid_stream;
75 tid_stream << std::this_thread::get_id();
76 std::string tid_str = tid_stream.str();
77
78 DLT_LOG(context, ConvertToDltLevel(level),
79 DLT_UINT32(static_cast<std::uint32_t>(pid));
80 DLT_CSTRING(tid_str.c_str()); DLT_CSTRING(name_.c_str());
81 DLT_SIZED_CSTRING(tag.data(), ConvertSizeToDltUint16(tag.size()));
82 DLT_SIZED_CSTRING(msg.data(), ConvertSizeToDltUint16(msg.size())));
83 }
84
85 void DoSetLogLevel(LogLevel level) override {
86 DLT_SET_APPLICATION_LL_TS_LIMIT(ConvertToDltLevel(level),
87 DLT_TRACE_STATUS_OFF);
88 }
89
90 private:
91 std::string name_ = "";
92 void CleanupDltResources() {
93 name_ = "";
94 for (auto& context_entry : context_map_) {
95 DLT_UNREGISTER_CONTEXT(*(context_entry.second));
96 }
97 context_map_.clear();
98 DLT_UNREGISTER_APP();
99 }
100
101 // Convert LogLevel to DltLogLevelType
102 static DltLogLevelType ConvertToDltLevel(LogLevel level) {
103 switch (level) {
104 case LogLevel::kFatal:
105 return DLT_LOG_FATAL;
106 case LogLevel::kError:
107 return DLT_LOG_ERROR;
108 case LogLevel::kWarn:
109 return DLT_LOG_WARN;
110 case LogLevel::kInfo:
111 return DLT_LOG_INFO;
112 case LogLevel::kDebug:
113 return DLT_LOG_DEBUG;
114 default:
115 return DLT_LOG_INFO;
116 }
117 }
118
119 // DLT requires context ID and APID to be max characters
120 static std::string ConvertToDltId(const std::string& input_id) {
121 // Return default ID for empty string as-is
122 if (input_id.empty()) {
123 return std::string(kDefaultDltID);
124 }
125 if (input_id.size() <= kDltIdMaxLength) {
126 return input_id;
127 }
128 return input_id.substr(0, kDltIdMaxLength);
129 }
130
131 DltContext& EnsureContext(const std::string& tag) {
132 const std::string context_id = ConvertToDltId(tag);
133 auto it = context_map_.find(context_id);
134 if (it != context_map_.end()) {
135 return *(it->second);
136 }
137
138 // Insert before registration to ensure cleanup on exception
139 auto context = std::make_unique<DltContext>();
140 auto emplace_result =
141 context_map_.try_emplace(context_id, std::move(context));
142 auto iter = emplace_result.first;
143 const bool inserted = emplace_result.second;
144 if (!inserted) {
145 return *(iter->second);
146 }
147
148 DLT_REGISTER_CONTEXT_LL_TS(
149 *(iter->second), context_id.c_str(), tag.c_str(),
150 static_cast<int32_t>(ConvertToDltLevel(GetLogLevelThreshold())),
151 DLT_TRACE_STATUS_OFF); // Full tag in description
152 return *(iter->second);
153 }
154
155 static std::uint16_t ConvertSizeToDltUint16(std::size_t value) {
156 constexpr std::size_t max_value =
157 static_cast<std::size_t>(std::numeric_limits<std::uint16_t>::max());
158 if (value > max_value) {
159 return std::numeric_limits<std::uint16_t>::max();
160 }
161 return static_cast<std::uint16_t>(value);
162 }
163
164 std::unordered_map<std::string, std::unique_ptr<DltContext>> context_map_;
165};
