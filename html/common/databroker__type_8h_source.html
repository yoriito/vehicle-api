343inline bool ValueTypeEquals(const ValueType& lhs, const ValueType& rhs) {
344 if (lhs.index() != rhs.index()) {
345 return false;
346 }
347
348 if (std::holds_alternative<StructDataPtr>(lhs)) {
349 const auto& left_ptr = std::get<StructDataPtr>(lhs);
350 const auto& right_ptr = std::get<StructDataPtr>(rhs);
351 if (!left_ptr || !right_ptr) {
352 return !left_ptr && !right_ptr;
353 }
354 return *left_ptr == *right_ptr;
355 }
356
357 if (std::holds_alternative<StructDataArray>(lhs)) {
358 const auto& left_array = std::get<StructDataArray>(lhs);
359 const auto& right_array = std::get<StructDataArray>(rhs);
360 if (left_array.size() != right_array.size()) {
361 return false;
362 }
363
364 const size_t array_size = left_array.size();
365 for (size_t i = 0; i < array_size; ++i) {
366 const auto& left_ptr = left_array[i];
367 const auto& right_ptr = right_array[i];
368 if (!left_ptr || !right_ptr) {
369 if (left_ptr || right_ptr) {
370 return false;
371 }
372 continue;
373 }
374
375 if (*left_ptr != *right_ptr) {
376 return false;
377 }
378 }
379
380 return true;
381 }
382
383 return lhs == rhs;
384}
