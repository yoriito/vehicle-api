145 if (!ret.IsOk()) {
146 std::unique_ptr<yoriito::ErrorType> err_ptr = ret.TakeErr();
147 YORIITO_LOG_ERROR("SAPI_SignalAPI", err_ptr->Message());
148 return yoriito::Result<yoriito::signal::SignalGetDataPoint<T>>::Err(

150 }
151 auto response = ret.TakeOk().value();
152 if (response.GetDataPoints().empty()) {
153 YORIITO_LOG_ERROR("SAPI_SignalAPI",
154 std::string(yoriito::ErrorType::to_msg(
155 yoriito::ErrorCode::ERROR_NODATA)));
156 return yoriito::Result<yoriito::signal::SignalGetDataPoint<T>>::Err(
157 std::make_unique<yoriito::CommonError>(
158 yoriito::ErrorCode::ERROR_NODATA));
159 }
160 const auto& data_point = response.GetDataPoints().front();
161 if constexpr (yoriito::signal::kIsCustomStruct<T>) {
162 const auto& data = data_point.GetData();
163 if (!std::holds_alternative<yoriito::StructDataPtr>(data)) {
164 YORIITO_LOG_ERROR("SAPI_SignalAPI", "Unexpected data type");
165 return yoriito::Result<yoriito::signal::SignalGetDataPoint<T>>::Err(
166 std::make_unique<yoriito::CommonError>(
167 yoriito::ErrorCode::ERROR_UNKNOWN));
168 }
169
170 const auto& struct_data_ptr = std::get<yoriito::StructDataPtr>(data);
171 if (!struct_data_ptr) {
172 YORIITO_LOG_ERROR("SAPI_SignalAPI", "Data pointer is null");
173 return yoriito::Result<yoriito::signal::SignalGetDataPoint<T>>::Err(
174 std::make_unique<yoriito::CommonError>(
175 yoriito::ErrorCode::ERROR_NODATA));
176 }
177
178 // Load struct data from StructData
179 T result;
180 if (!yoriito::types::internal::FromStructData(*struct_data_ptr, &result)) {
181 YORIITO_LOG_ERROR("SAPI_SignalAPI",
182 "Failed to extract {{ custom_struct.name | lower }} "
183 "data from StructData");
184 return yoriito::Result<yoriito::signal::SignalGetDataPoint<T>>::Err(
185 std::make_unique<yoriito::CommonError>(
186 yoriito::ErrorCode::ERROR_NODATA));
187 }
188 auto out = yoriito::signal::SignalGetDataPoint<T>(
189 data_point.GetPath(), result, data_point.IsAvailableSensor(),
190 data_point.IsAvailableActuator(), data_point.GetStoredTimestamp(),
191 data_point.GetProducedTimestamp());
192 YORIITO_LOG_DEBUG("SAPI_SignalAPI",
193 "Exiting function [SignalSubscribeContext.Messages].");
194 return yoriito::Result<yoriito::signal::SignalGetDataPoint<T>>::Ok(
195 yoriito::signal::SignalGetDataPoint<T>(out));
196 } else {
197 YORIITO_LOG_DEBUG("SAPI_SignalAPI",
198 "Exiting function [SignalSubscribeContext.Messages].");
199 return yoriito::Result<yoriito::signal::SignalGetDataPoint<T>>::Ok(
200 yoriito::signal::SignalGetDataPoint<T>(data_point));
201 }
